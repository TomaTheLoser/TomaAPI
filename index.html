Simple API
Clean, minimal API

No configuration files. No setup classes. Just call the method and it works.

// Full third-person player setup
public override void _Ready()
{
    CharacterController.Sprint = true;
    CharacterController.SprintSpeed = SprintSpeed;
}

public override void _PhysicsProcess(double delta)
{
    CharacterController.Update(this, Speed, JumpHeight, delta, camera, MovementStyle.General);
    TPP.Align(this, camera, 10.0f, delta, MovementStyle.General);
    TPP.EnableCollision(camera, this);
    TPP.Update(camera, this, delta: delta);
}
